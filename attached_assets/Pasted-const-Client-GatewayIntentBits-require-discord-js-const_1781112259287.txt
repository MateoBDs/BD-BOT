const { Client, GatewayIntentBits } = require('discord.js');

const client = new Client({
  intents: [
    GatewayIntentBits.Guilds,
    GatewayIntentBits.GuildMessages,
    GatewayIntentBits.MessageContent
  ]
});

client.once('ready', () => {
  console.log(`🤖 BD Bot conectado como ${client.user.tag}`);
});


// =========================
// 🧠 IA (BÁSICA / PLACEHOLDER)
// =========================
async function bdIA(prompt) {
  // Aquí luego puedes meter OpenAI o Groq API
  return `🤖 BD IA: He recibido tu mensaje: "${prompt}"`;
}


// =========================
// 🛍️ TIENDA BD
// =========================
const tienda = [
  { id: 1, nombre: "Web Básica", precio: 5 },
  { id: 2, nombre: "Web Pro", precio: 15 },
  { id: 3, nombre: "Bot Discord", precio: 10 }
];


// =========================
// 📦 STOCKS SEPARADOS
// =========================
let stockPedidos = {
  webs: 5,
  bots: 3
};

let stockTienda = {
  "Web Básica": 10,
  "Web Pro": 5,
  "Bot Discord": 3
};


// =========================
// 💬 COMANDOS
// =========================
client.on('messageCreate', async (message) => {
  if (message.author.bot) return;

  const msg = message.content;


  // =========================
  // 🤖 IA
  // =========================
  if (msg.startsWith('!ia ')) {
    const prompt = msg.slice(4);

    const respuesta = await bdIA(prompt);
    return message.reply(respuesta);
  }


  // =========================
  // 🛍️ TIENDA
  // =========================
  if (msg === '!tienda') {
    let texto = "🛍️ TIENDA BD:\n\n";

    tienda.forEach(p => {
      texto += `ID ${p.id} | ${p.nombre} - ${p.precio}€\n`;
    });

    return message.reply(texto);
  }


  // =========================
  // 📦 STOCK PEDIDOS
  // =========================
  if (msg === '!stock pedidos') {
    return message.reply(
      `📦 STOCK PEDIDOS:\n🌐 Webs: ${stockPedidos.webs}\n🤖 Bots: ${stockPedidos.bots}`
    );
  }


  // =========================
  // 📦 STOCK TIENDA
  // =========================
  if (msg === '!stock tienda') {
    let texto = "📦 STOCK TIENDA BD:\n\n";

    for (let item in stockTienda) {
      texto += `${item}: ${stockTienda[item]}\n`;
    }

    return message.reply(texto);
  }


  // =========================
  // 💰 COMPRAR TIENDA
  // =========================
  if (msg.startsWith('!buy ')) {
    const itemName = msg.slice(5);

    if (!stockTienda[itemName]) {
      return message.reply("❌ Ese producto no existe en la tienda");
    }

    if (stockTienda[itemName] <= 0) {
      return message.reply("❌ No hay stock de ese producto");
    }

    stockTienda[itemName]--;

    return message.reply(`✅ Compra realizada: ${itemName}\n📩 Contacta staff para entrega`);
  }


  // =========================
  // 🧪 PEDIDO WEB
  // =========================
  if (msg === '!buy web') {
    if (stockPedidos.webs <= 0) {
      return message.reply("❌ No hay stock de webs");
    }

    stockPedidos.webs--;

    return message.reply("✅ Pedido de web añadido a cola BD");
  }


  // =========================
  // 🔥 PING
  // =========================
  if (msg === '!ping') {
    return message.reply('🏓 Pong!');
  }
});
